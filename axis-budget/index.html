⚙️ One-time setup required

  1. Go to sheets.new to create a new Google Sheet.
  2. Click Extensions → Apps Script.
  3. Delete any existing code and paste the script below.
  4. Click Deploy → New Deployment → type: Web App → execute as Me → access Anyone.
  5. Copy the Web App URL and paste it as the SCRIPT_URL at the top of this file.
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const action = e.parameter.action || 'get';

  if (action === 'add') {
    const id = Date.now().toString(36) + Math.random().toString(36).slice(2,6);
    if (sheet.getLastRow() === 0) sheet.appendRow(['id','description','cost','notes','image']);
    sheet.appendRow([id, e.parameter.description||'', e.parameter.cost||'', e.parameter.notes||'', e.parameter.image||'']);
    return out({success:true, id});
  }

  if (action === 'edit') {
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] === e.parameter.id) {
        sheet.getRange(i+1,2,1,4).setValues([[
          e.parameter.description||'',
          e.parameter.cost||'',
          e.parameter.notes||'',
          e.parameter.image||''
        ]]);
        return out({success:true});
      }
    }
    return out({error:'not found'});
  }

  if (action === 'delete') {
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] === e.parameter.id) {
        sheet.deleteRow(i+1);
        return out({success:true});
      }
    }
    return out({error:'not found'});
  }

  // GET all items
  const rows = sheet.getDataRange().getValues();
  const items = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    items.push({id:rows[i][0], description:rows[i][1], cost:rows[i][2], notes:rows[i][3], image:rows[i][4]});
  }
  return out({items});
}

function out(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
