One-time setup: connect this app to your Google Sheet.

1
Create a new Google Sheet

Give it any name — e.g. "Ben's TO DO". Leave it blank; the app will create the right tabs automatically.

2
Open Apps Script

In the sheet: Extensions → Apps Script. Replace all existing code with the snippet below, then click Save.

function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); let todosSheet = ss.getSheetByName('Todos'); let catsSheet = ss.getSheetByName('Categories'); if (!todosSheet) { todosSheet = ss.insertSheet('Todos'); todosSheet.appendRow(['id','title','category_id','done','created','notes']); } if (!catsSheet) { catsSheet = ss.insertSheet('Categories'); catsSheet.appendRow(['id','name','color']); } function uid() { return Date.now().toString(36) + Math.random().toString(36).slice(2,6); } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } if (action === 'get') { const cd = catsSheet.getDataRange().getValues(); const cats = []; for (let i = 1; i < cd.length; i++) { if (cd[i][0]) cats.push({id:String(cd[i][0]),name:cd[i][1],color:cd[i][2]}); } const td = todosSheet.getDataRange().getValues(); const todos = []; for (let i = 1; i < td.length; i++) { if (td[i][0]) todos.push({ id:String(td[i][0]), title:td[i][1], category_id:String(td[i][2]||''), done: td[i][3]===true||td[i][3]==='TRUE', created:String(td[i][4]||''), notes:String(td[i][5]||'') }); } return json({todos, categories:cats}); } if (action === 'add_todo') { const id = uid(); todosSheet.appendRow([id, e.parameter.title||'', e.parameter.category_id||'', false, new Date().toISOString(), e.parameter.notes||'']); return json({success:true, id}); } if (action === 'edit_todo') { const td = todosSheet.getDataRange().getValues(); for (let i = 1; i < td.length; i++) { if (String(td[i][0]) === String(e.parameter.id)) { if (e.parameter.title !== undefined) todosSheet.getRange(i+1,2).setValue(e.parameter.title); if (e.parameter.category_id !== undefined) todosSheet.getRange(i+1,3).setValue(e.parameter.category_id); if (e.parameter.notes !== undefined) todosSheet.getRange(i+1,6).setValue(e.parameter.notes); break; } } return json({success:true}); } if (action === 'toggle_todo') { const td = todosSheet.getDataRange().getValues(); for (let i = 1; i < td.length; i++) { if (String(td[i][0]) === String(e.parameter.id)) { const cur = td[i][3]===true||td[i][3]==='TRUE'; todosSheet.getRange(i+1,4).setValue(!cur); break; } } return json({success:true}); } if (action === 'delete_todo') { const td = todosSheet.getDataRange().getValues(); for (let i = td.length-1; i >= 1; i--) { if (String(td[i][0]) === String(e.parameter.id)) { todosSheet.deleteRow(i+1); break; } } return json({success:true}); } if (action === 'add_category') { const id = uid(); catsSheet.appendRow([id, e.parameter.name||'', e.parameter.color||'#888888']); return json({success:true, id}); } if (action === 'edit_category') { const cd = catsSheet.getDataRange().getValues(); for (let i = 1; i < cd.length; i++) { if (String(cd[i][0]) === String(e.parameter.id)) { if (e.parameter.name) catsSheet.getRange(i+1,2).setValue(e.parameter.name); if (e.parameter.color) catsSheet.getRange(i+1,3).setValue(e.parameter.color); break; } } return json({success:true}); } if (action === 'delete_category') { const cd = catsSheet.getDataRange().getValues(); for (let i = cd.length-1; i >= 1; i--) { if (String(cd[i][0]) === String(e.parameter.id)) { catsSheet.deleteRow(i+1); break; } } return json({success:true}); } return json({error:'unknown action'}); }
